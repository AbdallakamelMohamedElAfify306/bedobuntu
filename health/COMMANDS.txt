# Mac commands for the cheetah health kit. One line each. Copy a line exactly.

## A. Connection and upload (once per boot / sleep)
# A1 bring the USB network up on the Mac (asks for your Mac password)
sudo ifconfig en6 inet 172.16.42.2 netmask 255.255.255.0 up
# A2 unzip the kit into your home folder (creates ~/ubuntu-health)
unzip -oq ~/Downloads/ubuntu-health.zip -d ~
# A3 upload the kit to the phone
scp -i "$CHEETAH_SSH_KEY" -r ~/ubuntu-health root@172.16.42.1:/root/
# A3-fallback if scp fails (no sftp)
tar -C ~ -cf - ubuntu-health | ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'tar -C /root -xf -'
# A4 verify the upload byte for byte
ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'cd /root/ubuntu-health && sha256sum -c SHA256SUMS'

## Step 0. Wallpaper override removal (idempotent)
# 0a check only
ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'bash /root/ubuntu-health/00-step0-undo-wallpaper.sh --check'
# 0b apply (restarts Phosh only if something was removed)
ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'bash /root/ubuntu-health/00-step0-undo-wallpaper.sh --apply'

## Phase 1. Read-only health check
# 1a full check (table on screen, full log on the phone)

# 1b lock screen inspection
ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'bash /root/ubuntu-health/11-lockscreen-check.sh --quiet'
# 1c lock screen fixes, dry run (pick the ones you need)

# 1d lock screen fixes, apply
ssh -i "$CHEETAH_SSH_KEY" root@172.16.42.1 'bash /root/ubuntu-health/12-lockscreen-fix.sh --apply tz clock24 autolock 120 notif-on keypad-fixed'
# 1e copy all logs to your Desktop (optional)
